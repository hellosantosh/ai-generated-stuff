6Code Examples

The brokerage publishes no SDK, so these examples build directly on each language's standard HTTP client. That is deliberate: it keeps the wire protocol visible, and it means nothing here goes stale when a vendor library does.

6.2Authorization: PKCE in three languages

The same four steps everywhere: generate the pair, build the authorization URL, verify state, exchange the code.

Python

import base64, hashlib, secrets

def generate_pkce_pair():
    """32 random bytes → a 43-character verifier, the RFC 7636 minimum."""
    verifier  = base64.urlsafe_b64encode(secrets.token_bytes(32)).decode().rstrip("=")
    digest    = hashlib.sha256(verifier.encode("ascii")).digest()
    challenge = base64.urlsafe_b64encode(digest).decode().rstrip("=")
    return verifier, challenge

# `secrets`, never `random` — the verifier is the only thing stopping
# someone who intercepts the authorization code from redeeming it.

tokens = exchange_code(code, verifier)   # POST /oauth2/token, no client_secret

Node.js

import { createHash, randomBytes, timingSafeEqual } from 'node:crypto';

export function generatePkcePair() {
  const codeVerifier  = randomBytes(32).toString('base64url');
  const codeChallenge = createHash('sha256').update(codeVerifier).digest('base64url');
  return { codeVerifier, codeChallenge };
}

// Constant-time compare on `state` — habits matter more than this one leak.
function safeEqual(a, b) {
  const x = Buffer.from(a), y = Buffer.from(b);
  return x.length === y.length && timingSafeEqual(x, y);
}

Java

public static PkcePair generatePkcePair() {
    byte[] entropy = new byte[32];
    RANDOM.nextBytes(entropy);                     // SecureRandom
    String verifier = b64url(entropy);

    byte[] digest = MessageDigest.getInstance("SHA-256")
            .digest(verifier.getBytes(StandardCharsets.US_ASCII));
    return new PkcePair(verifier, b64url(digest));
}

private static String b64url(byte[] raw) {
    return Base64.getUrlEncoder().withoutPadding().encodeToString(raw);
}
Refresh with rotation — all three clients share this shape

Persist the new refresh token before using the new access token, atomically (write-temp-then-rename). On invalid_grant, stop retrying and re-authorize: the token is revoked, expired, or already spent, and there is no other recovery. Serialize concurrent refreshes behind a lock, or two threads will trip reuse detection and revoke the family.

6.3Accounts: reading balances and positions

Note that all three follow the balances link rather than building the URL, and all three parse money into an exact decimal type.

Python

from decimal import Decimal

accounts = client.get("/accounts")["items"]
account  = accounts[0]

balances = client.follow(account, "balances")      # HATEOAS, not string concat

print(f"Net liquidation: {Decimal(balances['netLiquidation']):,.2f}")
print(f"Stock BP:        {Decimal(balances['buyingPower']['stock']):,.2f}")
print(f"Option BP:       {Decimal(balances['buyingPower']['option']):,.2f}")

# A margin call is the one balance field worth alerting on.
if (call := (balances.get("margin") or {}).get("marginCall")):
    alert(f"{call['type']}: {call['amount']} due {call['dueDate']}")

# paginate() follows `next` links; it never constructs a cursor.
total = sum(Decimal(p["unrealizedPnl"])
            for p in client.paginate(client.link(account, "positions")))

Node.js

const { items: accounts } = await client.get('/accounts');
const account  = accounts[0];
const balances = await client.follow(account, 'balances');

// Dec keeps values as strings and does exact arithmetic over BigInt.
// Number('412.50') would be fine; Number arithmetic would not be.
let totalUnrealized = '0';
for await (const p of client.paginate(client.link(account, 'positions'))) {
  totalUnrealized = Dec.add(totalUnrealized, p.unrealizedPnl);
}
console.log(`Unrealized: ${Dec.format(totalUnrealized)}`);

Java

JsonNode account  = client.get("/accounts").path("items").get(0);
JsonNode balances = client.follow(account, "balances");

System.out.printf("Net liquidation: %,14.2f%n", decimal(balances, "netLiquidation"));
System.out.printf("Stock BP:        %,14.2f%n", decimal(balances, "buyingPower", "stock"));

BigDecimal total = client.paginate(client.linkOrThrow(account, "positions"), Options.none())
        .map(p -> decimal(p, "unrealizedPnl"))
        .reduce(BigDecimal.ZERO, BigDecimal::add);

6.4Trading: placing a stock order

The pattern is identical across languages: generate the key once, preview, submit with the key, then confirm terminal state separately.

Python

order = {
    "assetClass":  "EQUITY",
    "side":        "BUY",
    "instrumentId": resolve_equity(client, "MSFT")["instrumentId"],
    "quantity":    "300",           # strings, never floats
    "orderType":   "LIMIT",
    "limitPrice":  "412.50",
    "timeInForce": "DAY",
}

result = client.post(f"{account_url}/orders/preview", order)
if not result["valid"]:
    for w in result["warnings"]:
        print(f"[{w['severity']}] {w['message']}")
    return

# Generated ONCE, outside the client's retry loop. Regenerating it in the
# loop is the exact bug idempotency exists to prevent.
key   = client.new_idempotency_key()
order = client.post(f"{account_url}/orders", order, idempotency_key=key)

print(order["orderId"], order["status"])   # NEW — routed, not filled

6.6Amend and cancel with optimistic concurrency

Read, check the affordance, mutate with If-Match, and handle 412 by re-reading rather than blind-retrying.

Python

def amend_order(client, order, new_price):
    self_url      = client.link(order, "self")
    fresh, headers = client.get_with_headers(self_url)

    # Branch on the link, not on the status string.
    if not client.can(fresh, "replace"):
        return fresh

    try:
        return client.patch(self_url, {"limitPrice": new_price},
                            if_match=headers.get("ETag"))
    except ApiError as exc:
        if exc.status != 412:
            raise
        # The order changed while we were deciding. Re-read and decide
        # again — never retry with the newer ETag unexamined.
        return client.get(self_url)

Node.js

const { body: fresh, headers } = await client.getWithHeaders(selfUrl);
if (!client.can(fresh, 'cancel')) return fresh;   // already terminal

const cancelled = await client.delete(selfUrl, { ifMatch: headers.get('ETag') });
// PENDING_CANCEL, not CANCELLED — the venue has not acknowledged yet.
console.log(cancelled.status);

Java

Response current = client.getWithHeaders(selfUrl, Options.none());
JsonNode fresh   = current.body();

if (!can(fresh, "replace")) return fresh;

try {
    ObjectNode patch = MAPPER.createObjectNode();
    patch.put("limitPrice", newPrice);
    return client.patch(selfUrl, patch, Options.none().ifMatch(current.headers().etag()));
} catch (ApiException e) {
    if (e.status() != 412) throw e;
    return client.get(selfUrl);      // re-read, then decide again
}
