Authorization with PKCE

The same four steps everywhere: generate the pair, build the URL, verify state, exchange the code.

import base64, hashlib, secrets

def generate_pkce_pair():
    """32 random bytes → a 43-character verifier, the RFC 7636 minimum."""
    verifier  = base64.urlsafe_b64encode(secrets.token_bytes(32)).decode().rstrip("=")
    digest    = hashlib.sha256(verifier.encode("ascii")).digest()
    challenge = base64.urlsafe_b64encode(digest).decode().rstrip("=")
    return verifier, challenge

# `secrets`, never `random` — the verifier is the only thing stopping
# someone who intercepts the code from redeeming it.

# Verify state in constant time before touching the code.
if not secrets.compare_digest(returned_state, expected_state):
    raise OAuthError("state_mismatch", "Possible CSRF; discarding the code.")

tokens = exchange_code(code, verifier)     # POST /oauth2/token, no client_secret
import { createHash, randomBytes, timingSafeEqual } from 'node:crypto';

export function generatePkcePair() {
  const codeVerifier  = randomBytes(32).toString('base64url');
  const codeChallenge = createHash('sha256').update(codeVerifier).digest('base64url');
  return { codeVerifier, codeChallenge };
}

// Constant-time compare on `state`.
function safeEqual(a = '', b = '') {
  const x = Buffer.from(a), y = Buffer.from(b);
  return x.length === y.length && timingSafeEqual(x, y);
}

const tokens = await exchangeCode(code, codeVerifier);
public static PkcePair generatePkcePair() {
    byte[] entropy = new byte[32];
    RANDOM.nextBytes(entropy);                     // SecureRandom
    String verifier = b64url(entropy);

    byte[] digest = MessageDigest.getInstance("SHA-256")
            .digest(verifier.getBytes(StandardCharsets.US_ASCII));
    return new PkcePair(verifier, b64url(digest));
}

private static String b64url(byte[] raw) {
    return Base64.getUrlEncoder().withoutPadding().encodeToString(raw);
}

// Constant-time compare on `state`.
if (!MessageDigest.isEqual(returned.getBytes(UTF_8), expected.getBytes(UTF_8))) {
    throw new OAuthException("state_mismatch", "Possible CSRF.");
}
Refresh with rotation — the shape all three share

Persist the new refresh token before using the new access token, and write it atomically. On invalid_grant, stop retrying and re-authorize — the token is revoked, expired, or already spent. Serialize concurrent refreshes behind a lock, or two threads will trip reuse detection and revoke the family.

Reading balances and positions

Note that all three follow the balances link rather than building the URL, and all three parse money into an exact decimal type. 

from decimal import Decimal
from brokerage_client import BrokerageClient

client   = BrokerageClient()
account  = client.get("/accounts")["items"][0]

# HATEOAS: follow the link, never concatenate the URL.
balances = client.follow(account, "balances")

print(f"Net liquidation: {Decimal(balances['netLiquidation']):,.2f}")
print(f"Stock BP:        {Decimal(balances['buyingPower']['stock']):,.2f}")
print(f"Option BP:       {Decimal(balances['buyingPower']['option']):,.2f}")

# A margin call is the one balance field worth alerting on.
if (call := (balances.get("margin") or {}).get("marginCall")):
    alert(f"{call['type']}: {call['amount']} due {call['dueDate']}")

# paginate() follows `next` links; it never constructs a cursor.
total = sum(Decimal(p["unrealizedPnl"])
            for p in client.paginate(client.link(account, "positions")))
print(f"Unrealized: {total:,.2f}")

Amend and cancel with optimistic concurrency

Read, check the affordance, mutate with If-Match, and handle 412 by re-reading rather than blind-retrying. 

def amend_order(client, order, new_price):
    self_url       = client.link(order, "self")
    fresh, headers = client.get_with_headers(self_url)

    # Branch on the link, not on the status string.
    if not client.can(fresh, "replace"):
        return fresh

    try:
        return client.patch(self_url, {"limitPrice": new_price},
                            if_match=headers.get("ETag"))
    except ApiError as exc:
        if exc.status != 412:
            raise
        # Changed while we were deciding. Re-read and decide again —
        # never retry with the newer ETag unexamined.
        return client.get(self_url)
export async function amendOrder(client, order, newPrice) {
  const selfUrl = client.link(order, 'self');
  const { body: fresh, headers } = await client.getWithHeaders(selfUrl);

  // Branch on the link, not on the status string.
  if (!client.can(fresh, 'replace')) return fresh;

  try {
    return await client.patch(selfUrl, { limitPrice: newPrice },
                                   { ifMatch: headers.get('ETag') });
  } catch (err) {
    if (!(err instanceof ApiError) || err.status !== 412) throw err;
    return client.get(selfUrl);   // re-read, then decide again
  }
}
Response current = client.getWithHeaders(selfUrl, Options.none());
JsonNode fresh   = current.body();

// Branch on the link, not on the status string.
if (!can(fresh, "replace")) return fresh;

try {
    ObjectNode patch = MAPPER.createObjectNode();
    patch.put("limitPrice", newPrice);
    return client.patch(selfUrl, patch,
                        Options.none().ifMatch(current.headers().etag()));
} catch (ApiException e) {
    if (e.status() != 412) throw e;
    return client.get(selfUrl);      // re-read, then decide again
}

Consuming the order event stream

The Node client ships a complete SSE consumer. It reconnects automatically, resumes from Last-Event-ID, skips heartbeat comments, and reassembles multi-line data: fields. 

const controller = new AbortController();
setTimeout(() => controller.abort(), 30_000);

const seen = new Set();
for await (const event of client.streamOrderEvents(account, { signal: controller.signal })) {
  // Delivery is at-least-once; dedupe on eventId.
  if (seen.has(event.eventId)) continue;
  seen.add(event.eventId);

  if (event.eventType === 'execution') {
    const fill = event.execution;
    console.log(`fill ${event.orderId}: ${fill.quantity} @ ${fill.price} → ${event.status}`);
  } else {
    console.log(`${event.eventType} ${event.orderId} → ${event.status}`);
  }
}
