   7import org.springframework.boot.SpringApplication;
   8import org.springframework.boot.autoconfigure.SpringBootApplication;
   9import org.springframework.boot.context.properties.ConfigurationProperties;
  10import org.springframework.boot.context.properties.EnableConfigurationProperties;
  11import org.springframework.context.annotation.Bean;
  12import org.springframework.security.oauth2.server.authorization.OAuth2TokenType;
  13import org.springframework.security.oauth2.server.authorization.token.JwtEncodingContext;
  14import org.springframework.security.oauth2.server.authorization.token.OAuth2TokenCustomizer;
  15
  16/**
  17 * A development-only OAuth 2.1 authorization server. Clients, scopes and the demo user are
  18 * declared in application.yaml; this class adds the two brokerage-specific claims that the
  19 * API relies on.
  20 */
  21@SpringBootApplication
  22@EnableConfigurationProperties(AuthServerApplication.Identities.class)
  23public class AuthServerApplication {
  24
  25    public static void main(String[] args) {
  26        SpringApplication.run(AuthServerApplication.class, args);
  27    }
  28
  29    /**
  30     * Who each token is for. A user signs in as themselves; a confidential client that uses
  31     * the client_credentials grant is registered to one customer, like a personal API key.
  32     * An optional per-client trading limit caps the notional value of any single order.
  33     */
  34    @ConfigurationProperties("brokerage.identities")
  35    record Identities(Map<String, String> customers, Map<String, BigDecimal> tradingLimits) {
  36        Identities {
  37            customers = customers == null ? Map.of() : customers;
  38            tradingLimits = tradingLimits == null ? Map.of() : tradingLimits;
  39        }
  40    }
  41
  42    /** Stamps the audience, customer_id and (if configured) trading_limit into every access token. */
  43    @Bean
  44    OAuth2TokenCustomizer<JwtEncodingContext> brokerageClaims(Identities identities) {
  45        return context -> {
  46            if (!OAuth2TokenType.ACCESS_TOKEN.equals(context.getTokenType())) {
  47                return;
  48            }
  49            // Tokens are minted for one API, which checks that it is the intended audience.
  50            context.getClaims().audience(List.of("brokerage-api"));
  51            // The principal is the signed-in user, or the client itself for client_credentials.
  52            String principal = context.getPrincipal().getName();
  53            String customerId = identities.customers().get(principal);
  54            if (customerId != null) {
  55                context.getClaims().claim("customer_id", customerId);
  56            }
  57            String clientId = context.getRegisteredClient().getClientId();
  58            BigDecimal limit = identities.tradingLimits().get(clientId);
  59            if (limit != null) {
  60                context.getClaims().claim("trading_limit", limit.toPlainString());
  61            }
  62        };
  63    }
  64}
